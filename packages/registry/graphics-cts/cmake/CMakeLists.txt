# Package-owned composition of the complete pinned upstream EGL/GLES2 modules.
# No test source lists, registration, assertions, result policies or filters are
# replaced. Vulkan/OpenGL/GLES3 suites are separate APIs, not selected here.
cmake_minimum_required(VERSION 3.20)
project(dEQP-Kandelo LANGUAGES C CXX)
set(DE_OS DE_OS_UNIX)
set(DE_CPU DE_CPU_WASM32)
set(DE_PTR_SIZE 4)
set(DE_DEBUG 1)
set(DEQP_TARGET_NAME "Kandelo GBM")
set(DEQP_ANDROID_APP OFF)
set(DEQP_ANDROID_EXE OFF)
set(DEQP_SUPPORT_GLES1 OFF)
set(DEQP_SUPPORT_VG OFF)
set(DEQP_SUPPORT_DRM ON)
set(DEQP_PLATFORM_LIBRARIES gbm EGL GLESv2 drm c++ c++abi)
set(DEQP_EGL_LIBRARIES EGL)
set(DEQP_GLES2_LIBRARIES GLESv2)
set(DETHREAD_STANDALONE_TEST OFF CACHE BOOL "Separate library self-test, not a CTS module")
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/framework/delibs/cmake")
include(framework/delibs/cmake/Defs.cmake NO_POLICY_SCOPE)
include(framework/delibs/cmake/CFlags.cmake)
add_compile_definitions(DEQP_TARGET_NAME="${DEQP_TARGET_NAME}" DE_ASSERT_FAILURE_CALLBACK DEQP_GLES2_DIRECT_LINK=1 DEQP_SUPPORT_DRM=1)
include(upstream-module-macros.cmake)
set(MODULE_LIB_TARGET_POSTFIX "-package")
set(MODULE_DATA_TARGET_POSTFIX "-data")

find_package(ZLIB REQUIRED)
find_package(PNG REQUIRED)
set(PNG_LIBRARY "${PNG_LIBRARIES}")
set(PNG_INCLUDE_PATH "${PNG_INCLUDE_DIRS}")
find_library(JSONCPP_LIBRARY NAMES jsoncpp REQUIRED)
find_path(JSONCPP_INCLUDE_DIR json/json.h REQUIRED)
add_library(jsoncpp_static STATIC IMPORTED)
set_target_properties(jsoncpp_static PROPERTIES IMPORTED_LOCATION "${JSONCPP_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${JSONCPP_INCLUDE_DIR}")
find_path(DRM_INCLUDE_DIR drm.h PATH_SUFFIXES drm libdrm REQUIRED)
find_path(DRM_PUBLIC_INCLUDE_DIR xf86drm.h PATH_SUFFIXES libdrm REQUIRED)
include_directories(${PNG_INCLUDE_DIRS} ${ZLIB_INCLUDE_DIRS} ${DRM_INCLUDE_DIR} ${DRM_PUBLIC_INCLUDE_DIR}
  framework/delibs/debase framework/delibs/depool framework/delibs/dethread
  framework/delibs/deutil framework/delibs/destream framework/delibs/decpp
  framework/common framework/qphelper framework/opengl framework/opengl/wrapper
  framework/referencerenderer framework/opengl/simplereference
  framework/randomshaders framework/egl framework/egl/wrapper framework/xexml)
foreach(lib debase depool dethread destream deutil decpp)
  add_subdirectory(framework/delibs/${lib})
endforeach()
add_subdirectory(framework/qphelper)
add_subdirectory(framework/common)
add_subdirectory(framework/opengl)
add_subdirectory(framework/egl)
add_subdirectory(framework/xexml)
add_subdirectory(framework/randomshaders)
add_subdirectory(framework/referencerenderer)
add_library(tcutil-platform STATIC framework/platform/tcuKandeloPlatform.cpp)
target_link_libraries(tcutil-platform tcutil glutil eglutil ${DEQP_PLATFORM_LIBRARIES})
add_subdirectory(modules/glshared)
add_subdirectory(modules/gles2)
add_subdirectory(modules/egl)
