cmake_minimum_required(VERSION 3.16)
project(v8_tracing_component_controls LANGUAGES CXX)
find_package(v8-tracing CONFIG REQUIRED)
add_executable(v8-tracing-component-controls component-controls.cpp)
target_compile_definitions(v8-tracing-component-controls PRIVATE _GNU_SOURCE)
target_link_libraries(v8-tracing-component-controls PRIVATE v8-tracing::v8-tracing)
add_library(v8-tracing-target-selectors OBJECT target-selectors.cpp)
target_link_libraries(v8-tracing-target-selectors PRIVATE v8-tracing::v8-tracing)
add_library(v8-tracing-legacy-target-selectors OBJECT target-selectors.cpp)
target_compile_definitions(v8-tracing-legacy-target-selectors PRIVATE KANDELO_TEST_LEGACY_OS_MACROS=1)
target_link_libraries(v8-tracing-legacy-target-selectors PRIVATE v8-tracing::v8-tracing)
